def find_median(arr1, arr2):
    n1 = len(arr1)
    n2 = len(arr2)
    if (n1 + n2) == 0:
        return None

    i, j = 0, 0
    last_1, last_2 = None, None

    for k in range(1, (n1 + n2) // 2 + 2):
        last_2 = last_1
        if j == n2:
            last_1 = arr1[i]
            i += 1
        elif i == n1:
            last_1 = arr2[j]
            j += 1
        elif arr1[i] < arr2[j]:
            last_1 = arr1[i]
            i += 1
        else:
            last_1 = arr2[j]
            j += 1
        
    if (n1 + n2) % 2 == 1:
        return last_1
    else:
        return (last_1 + last_2) / 2
