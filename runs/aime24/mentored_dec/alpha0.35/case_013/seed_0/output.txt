\[
\begin{aligned}
(75+117i)z

\frac{96+144i}{z}
    &= \frac{96+144i}{4}e^{-i\theta}
    = (24+36i)(\cos\theta-i\sin\theta).
\end{aligned}
\]

\[
\begin{aligned}
\operatorname{Re}\bigl((75+117i)z+\frac{96+144i}{z}\bigr)

   + (24\cos\theta+36\sin\theta) \\
 &= (300+24)\cos\theta+(-468+36)\sin\theta \\

\end{aligned}
\]
