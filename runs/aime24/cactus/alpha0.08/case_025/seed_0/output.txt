\[
\begin{cases}
a-b-c = \tfrac12 \\
-b+a-c = \tfrac13 \\
-c-a-b = \tfrac14
\end{cases}
\]

Add the first two equations:

\[
a-b-c -a+b-c = \tfrac12+\tfrac13 \quad\Longrightarrow\quad -2c = \tfrac56\quad\Longrightarrow\quad c=-\tfrac5{12}.
\]

Add the first and the third:

\[
a-b-c-a-b+c = \tfrac12+\tfrac14 \quad\Longrightarrow\quad -2b = \tfrac34\quad\Longrightarrow\quad b=-\tfrac38.
\]

Add the last two:

\[
-b+a-c-b-a+c = \tfrac13+\tfrac14 \quad\Longrightarrow\quad -2a = \tfrac7{12}\quad\Longrightarrow\quad a=-\tfrac7{24}.
\]

So  

\[
a=-\frac{7}{24},\qquad b=-\frac{9}{24},\qquad c=-\frac{10}{24}.
\]

\[
\log_2(x^4y^3z^2)=4a+3b+2c.

\[

\begin{aligned}
