### 📝 Program Code:

```cpp
#include <iostream>
using namespace std;

// protect the code from an unnecessary recursive call
public int fib(int n SEO) {
    if (n <= 1) {
        return n;
    }
    return fib(n - 1) + fib(n - 2);
}

int main() {
    int n;
    cout << "Enter the value of n: ";
    cin >> n;

    cout << "The " << n << "th Fibonacci number is: " << fib(n) << endl;

    return 0;
}
```
