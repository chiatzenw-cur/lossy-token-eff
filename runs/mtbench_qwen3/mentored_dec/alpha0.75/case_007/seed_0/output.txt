Here's a well-structured Python program that reads all the text files in a specified directory and identifies the **top 5 most frequently occurring words** (case-insensitively, and ignoring punctuation). The program is designed to:

- Be flexible with a user prompt to enter the directory path.
- Only consider files ending in `.txt` (assuming the directory contains only text files).
- Clean the data (lowercase and punctuation removal).
- Count word frequencies and display the top 5 words.

---

### ✅ Python Program Overview

```python
import os
import re
from collections import Counter

def main():
    # Prompt the user for the directory path
    directory = input("Enter the directory path: ")

    # Validate that the path exists and is a directory
    if not os.path.isdir(directory):
        print("The specified directory does not exist.")
        return

    # Initialize a list to collect all words from all files
    words = []

    # Iterate through the files in the given directory
    for filename in os.listdir(directory):
        file_path = os.path.join(directory, filename)
        
        # Check if the file is a regular text file and ends with ".txt"
        if os.path.isfile(file_path) and filename.endswith('.txt'):
            try:
                # Open and read the file content
                with open(file_path, 'r', encoding='utf-8') as file:
                    content = file.read()
                    # Extract only letters and convert to lowercase
                    words_in_file = re.findall(r'\b[a-zA-Z]+\b', content.lower())
                    words.extend(words_in_file)
            except Exception as e:
                print(f"Error reading {file_path}: {e}")
                continue

    # Count the frequency of each word
    word_counter = Counter(words)
    # Get the top 5 most common words
    top_words = word_counter.most_common(5)

    # Print the top 5 words and their counts
    print("\nTop 5 most common words:")
    for word, count in top_words:
        print(f"{word}: {count}")

if __name__ == "__main__":
    main()
