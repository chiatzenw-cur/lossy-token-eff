Here's a C++ program that calculates the nth Fibonacci number, following the standard definition of the Fibonacci sequence where $ F(0) = 0 $, $ F(1) = 1 $, and $ F(n) = F(n-1) + F(n-2) $ for $ n \geq 2 $. The program uses recursion and includes a simple input validation to handle edge cases.

---

### ✅ C++ Program to Find the Nth Fibonacci Number using Recursion

```cpp
#include <iostream>
using namespace std;

// Recursive function to calculate the Fibonacci number fib(n)
int fib(int n) {
    if (n == 0) {
        return n;
    }
    if (n == 1) {
        return n;
    }
    return fib(n - 1) + fib(n - 2);
}

int main() {
    int n;

    // Prompt the user to enter a number
    cout << "Enter the value of n: ";
    cin >> n;

    // Check if n is valid (n >= 0)
    if (n < 0) {
        cout << "Invalid input. n cannot be negative." << endl;
        return 1;  // Return Abnormal
    }

    // Calculate and display Fibonacci number
    int result = fib(n);
    cout << "The 'nth' Fibonacci number is: " << result << endl;

    return 0;
}
```
