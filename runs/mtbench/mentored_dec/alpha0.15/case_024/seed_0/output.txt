We have two integer arrays `A` and `B`.  
We have to print all values that appear in **both** arrays but we are not allowed to use any
auxiliary data structures such as hashing or additional vectors.

The easiest way to do this without external storage is:

1. **Sort both arrays in place** – this only requires the memory the arrays already occupy.
2. Scan the two sorted arrays simultaneously with two indices  
   (`i` for `A`, `j` for `B`).  
   - If `A[i] == B[j]` we found a common element → output it and advance both pointers.  
   - If `A[i] < B[j]` we need a larger value in `A` → advance `i`.  
   - Otherwise (`A[i] > B[j]`) → advance `j`.

This algorithm needs no auxiliary data structures: only the two index variables.

The time complexity is  

```
sorting A  -> O(n1 log n1)
sorting B  -> O(n2 log n2)
two‑pointer scan -> O(n1 + n2)
```

So overall `O(n1 log n1 + n2 log n2)` time,  
and `O(1)` extra space (apart from the input arrays).

### C++17 Implementation

```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n1, n2;
    if (!(cin >> n1)) return 0;          // no input
    vector<int> A(n1);
    for (int i = 0; i < n1; ++i) cin >> A[i];

    cin >> n2;
    vector<int> B(n2);
    for (int i = 0; i < n2; ++i) cin >> B[i];

    sort(A.begin(), A.end());            // sort in place
    sort(B.begin(), B.end());

    int i = 0, j = 0;
    vector<int> common;                  // used only to detect “no common”
    while (i < n1 && j < n2) {
        if (A[i] == B[j]) {
            common.push_back(A[i]);      // store for output
            ++i; ++j;
        } else if (A[i] < B[j]) {
            ++i;
        } else {
            ++j;
        }
    }

    if (common.empty()) {
        cout << "No common elements\n";
    } else {
        for (size_t k = 0; k < common.size(); ++k) {
            if (k) cout << ' ';
            cout << common[k];
        }
        cout << '\n';
    }
    return 0;
}
```
