```python
#!/usr/bin/env python3
"""
Word Counter – Top‑5 Word Frequency

Usage
-----
    python wordcounter.py  /path/to/directory   # prints the top‑5 words

Features
--------
*  Recursively scans a directory tree for .txt files.
*  Handles UTF‑8 encoding (and falls back to UTF‑8 with errors='ignore').
*  Normalises text to lower‑case, removes punctuation (digits & letters only).
*  Counts words with a collections.Counter and reports the top‑5.

Author: OpenAI ChatGPT
Date: 2026‑08‑01
"""

import argparse
import re
import sys
from collections import Counter
from pathlib import Path


def parse_args() -> argparse.Namespace:
    """Parse command‑line arguments."""
    parser = argparse.ArgumentParser(
        description="Count words in all text files under a directory and "
        "report the top‑5 most frequent words."
    )
    parser.add_argument(
        "directory",
        type=Path,
        help="Path to the directory containing the text files."
    )
    parser.add_argument(
        "-e",
        "--exclude",
        nargs='*',
        default=[],
        help="File patterns to skip (e.g., *.log *.txt)."
    )
    parser.add_argument(
        "-s",
        "--stopwords",
        action="store_true",
        help="Ignore common English stop‑words (the, a, of, …)."
    )
    return parser.parse_args()


def get_text_files(root: Path, exclude: list[str]) -> list[Path]:
    """
    Recursively find all .txt files under *root*,
    but skip those that match any of the patterns in *exclude*.
    """
    # Use the internal glob pattern; Path.glob works like '**/*.txt' recursively.
    pattern = '**/*.txt'
    files = list(root.rglob(pattern))

    if exclude:
        # Convert patterns to simple glob patterns
        excluded = []
        for pat in exclude:
            excluded.extend(root.rglob(pat))
        excluded = set(excluded)

        files = [f for f in files if f not in excluded]

    return files


def tokenize(text: str) -> list[str]:
    """
    Return a list of lowercase words extracted from *text*.
    Only alphanumeric characters are considered; punctuation is ignored.
    """
    # Example: "Hello, world!" -> ["hello", "world"]
    return re.findall(r'[A-Za-z0-9]+', text.lower())


def read_file(path: Path) -> str:
    """
    Safely read a text file; try UTF‑8 with different error handling.
    """
    try:
        return path.read_text(encoding="utf-8")
    except UnicodeDecodeError:
        # Fall back to "replace" or "ignore" modes for safety
        return path.read_text(encoding="utf-8", errors="replace")


def load_stopwords() -> set[str]:
    """
    Return a set of common English stop‑words. The list is limited
    to keep the script lightweight – feel free to expand it.
    """
    return {
        "the", "and", "a", "an", "in", "to", "of", "that", "for",
        "with", "on", "as", "by", "at", "it", "is", "was", "were",
        "be", "been", "this", "are", "from", "or", "have", "has",
        "but", "not", "they", "their", "you", "we",
    }


def main() -> None:
    args = parse_args()

    if not args.directory.exists():
        sys.exit(f"❌ Directory not found: {args.directory}")

    files = get_text_files(args.directory, args.exclude)
    if not files:
        sys.exit(f"🚫 No .txt files found in {args.directory}")

    counter = Counter()
    for file in files:
        text = read_file(file)
        words = tokenize(text)
        if args.stopwords:
            words = [w for w in words if w not in load_stopwords()]
        counter.update(words)

    if not counter:
        sys.exit("⚠️  No words found after processing the files.")

    # Take the top‑5 items
    top5 = counter.most_common(5)

    # Pretty‑print the result
    print("\n=== Top‑5 Words ===")
    for word, freq in top5:
        print(f"{word:<15} {freq} ×")
    print("===================\n")

    # Optionally, show the total number of distinct words counted
    total_words = sum(counter.values())
    distinct_words = len(counter)
    print(f"Total word instances found: {total_words:,}")
    print(f"Distinct words counted: {distinct_words:,}")


if __name__ == "__main__":
    main()
```

## How the script works

| Step | What happens | Why it matters |
|------|--------------|----------------|
| **CLI arguments** | `directory` argument, optional `-e` (exclude patterns), `-s` (stop‑words) | Lets you target any folder and maintain clean output |
| **File discovery** | `Path.rglob("**/*.txt")` | Recursively collects all `.txt` files |
| **Tokenisation** | `re.findall(r'[A-Za-z0-9]+', text.lower())` | Keeps only alphabetic/digit parts, folds case, replaces punctuation |
| **Optional stop‑word filtering** | `load_stopwords()` | Excludes common English words, useful for more meaningful stats |
| **Counting** | `Counter.update()` | Efficient word frequency tally |
| **Reporting** | `most_common(5)` | Prints the 5 most common words with their counts |

## Example usage

Suppose you have a directory `/tmp/my_texts` with a handful of documents:

```bash
/tmp/my_texts/
├── f1.txt
├── f2.txt
└── subdir/
    └── f3.txt
```

Run:

```bash
python wordcounter.py /tmp/my_texts
```

You should see output like:

```
=== Top‑5 Words ===
the              874 ×
and              457 ×
to               312 ×
of                299 ×
in                255 ×
===================

Total word instances found: 3,452,000
Distinct words counted: 12,538
```

If you want to drop the stop‑words:

```bash
python wordcounter.py /tmp/my_texts --stopwords
```

If you also want to ignore a few file patterns such as a log:

```bash
python wordcounter.py /tmp/my_texts -e "*.log" -s
```
