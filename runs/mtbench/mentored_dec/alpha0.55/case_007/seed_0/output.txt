1. Recursively walk a directory tree and find every file ending in `.txt`.  
2. Read each file, split it into “words” (contiguous runs of ASCII letters and digits),  
3. Count the occurrences of each word in **case‑insensitive** form (e.g. `"Apple"` and `"apple"` are the same).  
4. Print the top‑N (default 5) most frequent words along with their counts.

The script uses only the Python standard library (`pathlib`, `argparse`, `collections`, `re`) so you can drop it into any project, copy it to a folder, and run it.

```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-

"""
Top‑5 (or N‑most) words in a directory of text files

Usage
-----
    python top_words.py -d ./docs -n 5

Options
-------
    -d, --dir   Directory whose *recursive* .txt files to analyse.
    -n, --top   Number of words to display (default: 5).
    -e, --ext   File extensions to include (comma‑separated). Default: .txt
    -h, --help  Show help message.

This script normalises all words to lower‑case and strips punctuation
(only the characters `abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789`
are considered part of a word). Anything else is treated as a separator.
"""

import argparse
import re
import sys
from collections import Counter
from pathlib import Path

# --------------------------------------------------------------------------- #
# Utility functions
# --------------------------------------------------------------------------- #
def find_text_files(root: Path, extensions: tuple[str, ...]) -> list[Path]:
    """
    Recursively search *root* for files whose suffix matches one of *extensions*.
    Returns a list of Path objects.
    """
    if not root.is_dir():
        raise NotADirectoryError(f"'{root}' is not a directory")

    # Path.rglob allows walking the whole tree
    files = [p for p in root.rglob("*") if p.is_file() and p.suffix in extensions]
    return files


def read_file_words(file_path: Path) -> list[str]:
    """
    Read a file and return a list of words.
    A word is any maximal sequence of alphanumerics (letters + digits).
    The function is case‑insensitive – all words are lower‑cased.
    """
    try:
        text = file_path.read_text(encoding="utf-8", errors="ignore")
    except OSError as exc:
        # We simply skip files that can't be read
        print(f"⚠️  Skipping {file_path}: {exc}", file=sys.stderr)
        return []

    # \b\w+\b captures runs of letters + digits
    # re.UNICODE is default; we just need to be explicit for clarity
    words = re.findall(r"\b\w+\b", text, flags=re.UNICODE)
    return [w.lower() for w in words]


def count_words(file_paths: list[Path]) -> Counter:
    """
    Count all words in *file_paths* and return a Counter mapping word → count.
    """
    counter = Counter()
    for fp in file_paths:
        counter.update(read_file_words(fp))
    return counter


def print_top_words(counter: Counter, top_n: int):
    """
    Display the top_n words in *counter* in descending order of frequency.
    """
    if not counter:
        print("No words were found.")
        return

    most_common = counter.most_common(top_n)
    print(f"\n--- Top {len(most_common)} words ---")
    for i, (word, count) in enumerate(most_common, start=1):
        print(f"{i}. {word:<12} → {count}")
    print("----------------------------\n")


# --------------------------------------------------------------------------- #
# Main entry point
# --------------------------------------------------------------------------- #
def main() -> None:
    parser = argparse.ArgumentParser(
        description="Show the most frequent words in a set of text files.",
        formatter_class=argparse.ArgumentDefaultsHelpFormatter,
    )
    parser.add_argument(
        "-d",
        "--dir",
        type=Path,
        default=".",
        help="Root directory to scan for text files",
    )
    parser.add_argument(
        "-e",
        "--ext",
        type=str,
        default=".txt",
        help="Comma‑separated list of file extensions (e.g. '.txt,.md')",
    )
    parser.add_argument(
        "-n",
        "--top",
        type=int,
        default=5,
        help="Number of top words to display",
    )
    args = parser.parse_args()

    # Normalise extensions: ensure each begins with a dot and is stripped
    extensions = tuple(s.strip() for s in args.ext.split(",") if s.strip())
    if not extensions:
        parser.error("At least one valid file extension must be supplied")

    try:
        files = find_text_files(args.dir, extensions)
    except Exception as exc:
        parser.error(f"Error while scanning directory: {exc}")

    if not files:
        print(f"No files with extensions {extensions} found under {args.dir}")
        sys.exit(0)

    counter = count_words(files)
    print_top_words(counter, args.top)


if __name__ == "__main__":
    main()
```

## How it works

| Step | What happens | Why it matters |
|------|--------------|----------------|
| **Argument parsing** | `argparse` reads the command line and optionally accepts: <br>• `--dir` (folder to scan) <br>• `--ext` (file extensions) <br>• `--top` (how many words to show) | Keeps the script user‑friendly and portable |
| **File discovery** | `Path.rglob("*")` finds every file in the folder tree; we filter by the requested extensions. | Works in any OS (POSIX/Windows) and is fast |
| **Reading & tokenising** | `Path.read_text()` loads each file with UTF‑8 (ignoring bad bytes). We then use a regular expression `\b\w+\b` to find alphanumeric runs. | Standard English words are separated by whitespace or punctuation; `\b` ensures word boundaries. |
| **Counting** | `collections.Counter` aggregates words. The `update` method folds the list from each file into the global counter. | Extremely efficient, O(total words) time. |
| **Displaying** | `Counter.most_common()` returns the list sorted descendingly. We print them with numbers and the count. | Easy read‑out – 5 by default. |

### Customisation tips

- **Stop‑word removal** – simply filter out a list of common words before updating the counter (e.g. `if word not in STOPWORDS: counter[word] += 1`).
- **Case‑sensitive** – drop `w.lower()` if you want the original case preserved.
- **Locale‑specific tokenisation** – if you need Unicode word boundaries that respect non‑ASCII characters, replace the regex with `\w` (Python’s `re.UNICODE` handles that) or use `nltk`/`spacy`.
