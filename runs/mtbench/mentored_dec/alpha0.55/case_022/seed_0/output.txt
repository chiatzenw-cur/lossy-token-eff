```cpp
// FibonacciRecursion.cpp
// Compile with:  g++ -std=c++17 -O2 FibonacciRecursion.cpp -o fib
#include <iostream>
#include <stdexcept>

using std::cout;
using std::cin;
using std::endl;

/*
 * Recursive function that returns the nth Fibonacci number.
 * The function follows the classic definition:
 *   Fib(0) = 0
 *   Fib(1) = 1
 *   Fib(n) = Fib(n-1) + Fib(n-2)  for n >= 2
 *
 * The return type is long long because the values grow quickly
 * and a 32‑bit int will overflow well before n = 47.
 */
long long fib(int n)
{
    if (n < 0) {
        throw std::invalid_argument("n must be non‑negative");
    }

    if (n <= 1)            // base case
        return n;

    // recursive call
    return fib(n-1) + fib(n-2);
}

int main()
{
    int n;
    cout << "Enter n (non‑negative integer): ";
    if(!(cin >> n)) {
        cerr << "Invalid input!" << endl;
        return 1;
    }

    try {
        cout << "Fib(" << n << ") = " << fib(n) << endl;
    } catch (const std::exception &e) {
        cerr << "Error: " << e.what() << endl;
        return 1;
    }

    return 0;
}
```

### How it works

1. **Base case** – When `n` is 0 or 1 we return `n` directly.  
2. **Recursive step** – For any other `n`, we compute `fib(n-1)` and `fib(n-2)` recursively, adding the results.  
3. **Main** – Reads `n` from the user, calls `fib`, and prints the result. If the input is negative, an exception is thrown.

> **Note** – The naïve recursive approach has exponential time complexity (O(2ⁿ)) and is only practical for small `n` (≤ 40‑45).  
> For larger `n` you’d want to use memoisation, dynamic programming, or an iterative approach.
