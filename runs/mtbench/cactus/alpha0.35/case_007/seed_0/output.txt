#!/usr/bin/env python3
"""
top5_word_counter.py

Counts the most frequent words in all .txt files inside a directory.
Usage:   python top5_word_counter.py /path/to/directory
"""

from __future__ import annotations

import argparse
import sys
import os
import re
from collections import Counter, defaultdict
from pathlib import Path
from typing import Iterable

# --------------------------------------------------------------------------- #
# Helper functions
# --------------------------------------------------------------------------- #

def strip_punctuation(text: str) -> str:
    """
    Remove punctuation and convert to lower‑case.
    """
    # Use regex to replace every non‑alphanumeric (alpha‑num and apostrophes) with a space
    return re.sub(r"[^\w']+", " ", text.lower())

def preprocess(text: str) -> Iterable[str]:
    """
    Turn a string into a list of the words cleaned with punctuation removed.
    """
    # `strip_punctuation` has already replaced punctuation with space.
    # So we just split on whitespace.
    return strip_punctuation(text).split()

def word_counter_from_file(path: Path) -> Counter[str]:
    """
    Read a single txt file and return a Counter of the words.
    """
    try:
        with path.open(encoding="utf-8") as fp:
            return Counter(preprocess(fp.read()))
    except Exception:                            # pragma: no cover – catches encoding issues
        sys.stderr.write(f"⚠️  Could not read {path}: {sys.exc_info()[1]}\n")
        return Counter()

def all_text_files(directory: Path) -> Iterable[Path]:
    """
    Recursively yield all *.txt files in *directory* (case‑insensitive).
    """
    for root, _, files in os.walk(directory):
        for name in files:
            if name.lower().endswith(".txt"):
                yield Path(root) / name

# --------------------------------------------------------------------------- #
# Core logic
# --------------------------------------------------------------------------- #

def gather_phrases(root: Path, preprocess_fn: callable = preprocess) -> Counter:
    """
    Walk the directory tree, read all txt files, accumulate counters.
    """
    overall_counter = Counter()
    for txt_path in all_text_files(root):
        overall_counter.update(word_counter_from_file(txt_path))
    return overall_counter


def main() -> None:
    parser = argparse.ArgumentParser(
        description="Count words in all .txt files in a directory and print the top‑5."
    )
    parser.add_argument(
        "directory",
        type=Path,
        help="Path to the directory that contains the text files.",
    )
    args = parser.parse_args()

    if not args.directory.is_dir():

    counter = gather_phrases(args.directory)

    if not counter:

    print(f"📊  Total words found: {total_tokens}")
    print(f"🗂️  Total distinct words: {distinct_words}")

    # Show top‑5
    top5 = counter.most_common(5)
    print("\n  ⬆️  Top‑5 MOST FREQ‑UENT words (all Lower‑case, punctuation removed):")
