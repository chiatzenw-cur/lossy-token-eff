```cpp
// fib.cpp - Recursive Fibonacci number demo
// Compile with: g++ -std=c++17 -O2 fib.cpp -o fib

#include <iostream>
#include <limits>

using namespace std;

// ---------------------------------------------------------------------
// Recursive Fibonacci function
// ---------------------------------------------------------------------
long long fib(int n)
{
    // very small combinatorial limits, but feel free to add validation
    if (n == 0)   // Base case – 0th Fibonacci number
        return 0;
    if (n == 1)   // Base case – 1st Fibonacci number
        return 1;

    // Recursive case
    return fib(n - 1) + fib(n - 2);
}

// ---------------------------------------------------------------------
// Main – read & print the value
// ---------------------------------------------------------------------
int main()
{
    int n;
    cout << "Enter a non‑negative integer: ";
    while (!(cin >> n) || n < 0) {
        // Reset min input stream if anything fails
        cin.clear();
        cin.ignore(numeric_limits<streamsize>::max(), '\n');
        cout << "Invalid input. Please enter a non‑negative integer: ";
    }

    cout << "Fibonacci number F(" << n << ") = " << fib(n) << '\n';
    return 0;
}
```

### What the code does

1. **Input handling** – The program collects a non‑negative integer `n` from the user.  
2. **Recursive function** – `fib(n)` returns `0` if `n` is `0`, `1` if `n` is `1`, and otherwise
   it calls itself twice to compute the two preceding terms and sums them.  
3. **Output** – `main` prints the result and terminates.

> **Note**  
> This straightforward implementation is *exponential* in time – O(1.618ⁿ) – for large `n`.  
> For production code, move to an iterative or memo‑ized version to avoid exponential blow‑up.
