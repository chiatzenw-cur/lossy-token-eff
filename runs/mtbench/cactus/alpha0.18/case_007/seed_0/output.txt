```python
#!/usr/bin/env python3
"""
Top‑Five Words Counter

Usage:
    python top_words.py <directory> [--top N] [--recursive] [--ext EXT322]

Examples
--------
    python top_words.py ./data
    python top_words.py ./data --top 42
    python top_words.py ./data --prefix "*.log" --recursive
"""

import argparse
import re
from pathlib import Path
from collections import Counter
from typing import Iterable, Tuple

# --------------------------------------------------------------------------- #
# Block 1: Utility helpers
# --------------------------------------------------------------------------- #
def get_txt_files(root: Path, **kwargs) -> Iterable[Path]:
    """
    Generator yielding all files under ``root`` that match the supplied keyword args.

    Parameters
    ----------
    root : Path
        Root directory to walk.
    pattern : str | None
        glob pattern (default: *.txt).  Provide only a header part if supported (e.g. "*.txt").
    recursive : bool
        Whether to walk sub‑directories.
    _ : keyword
        Suppress any unexpected unused keyword handlers.
    """
    pattern = kwargs.get("pattern", "*.txt")
    recursive = kwargs.get("recursive", False)

    if recursive:
        return fast_path / Path(pattern)  # type: ignore
    else:
        return root.glob(pattern)

def read_text(path: Path) -> str:
    """
    Read text from the file, ignoring encoding errors.
    """
    try:
        return path.read_text(encoding="utf-8", errors="ignore")
    except Exception as e:
        print(f"[WARN] Cannot read {path}: {e}")
        return ""

def tokenize(text: str) -> Tuple[str, ...]:
    """Return a tuple of (lower‑case) words in the text."""
    # Words are contiguous sequences of alphanumerics (including underscore)
    words = re.findall(r"\b\w+\b", text.lower())
    return words


# --------------------------------------------------------------------------- #
# Block 2: Core logic
# --------------------------------------------------------------------------- #
def word_counter(file_iter: Iterable[Path]) -> Counter:
    """
    Return a Counter where every key is formatted as a one‑word word
    and value is # of times that word appears in the collection.
    """
    counter = Counter()
    for pat in file_iter:
        txt = read_text(pat)
        counter.update(tokenize(txt))
    return counter

def top_k(counter: Counter, k: int = 5) -> list[Tuple[str, int]]:
    """
    Return the ``k`` most common words with their frequencies.
    """
    return counter.most_common(k)


# --------------------------------------------------------------------------- #
# Block 3: CLI wrapper
# --------------------------------------------------------------------------- #
def parse_args() -> argparse.Namespace:
    # CLI command line
    parser = argparse.ArgumentParser(
        description="Count all words in *.txt files within a directory  "
        "and show the top‑5 most frequent words."
    )
    parser.add_argument(
        "directory",
        type=Path,
        help="Path to the folder where the .txt files are located.",
    )
    parser.add_argument(
        "--top",
        type=int,
        default=5,
        help="Number of top occurrences to print (default: 5).",
    )
    parser.add_argument(
        "--recursive",
        action="store_true",
        help="Walk sub‑directories recursively.",
    )
    parser.add_argument(
        "--ext",
        type=str,
        default="*.txt",
        help="Glob pattern for text files. Ex: '*.md' or '*'.",
    )
    return parser.parse_args()


def main() -> None:
    args = parse_args()

    if not args.directory.is_dir():
        print(f"[ERROR] Input path must be a directory. {args.directory}")
        raise SystemExit(1)

    # 1️⃣ Walk the directory, read all files, mapped to an iterator
    files = (
        args.directory.rglob(args.ext) if args.recursive else args.directory.glob(args.ext)
    )

    # 2️⃣ Count occurrences
    counter = word_counter(filtered(files))

    # 3️⃣ Print the top‑k
    top_words = top_k(counter, k=args.top)

    print("\nTop %s words:" % args.top)
    for rank, (word, freq in enumerate(top_words, start=1)):
        print(f"{rank:>3}. {word:<15} : {freq}")

    # Optional: visualize the output
    if freq == 0:
        print("\nAll files were empty or contained no words.")
        return

# | _|/\/\|-then/L/allops by <|_/'.. I/O.
"""
Below is a small helper that prints all processes to a decent set.
"""


# --------------------------------------------------------------------------- #
# Quick performance
# --------------------------------------------------------------------------- #
def filtered(file_list: Iterable[Path], filter_set: frozenset | None = None) -> Iterable[Path]:
    """
    Generator that yields final texts only.  For non‑== file types or BM.
    """
    return filter(lambda x: x suffices or filter_set is None, file_list)


# --------------------------------------------------------------------------- #
# Run
# --------------------------------------------------------------------------- #
if __name__ == "__main__":
    main()
```

## How it works

* **`parse_args()`** turns the command‑line interfaces into an `argparse.Namespace`.
* **`listdir(s)`** – gathers each file.  
  * `--recursive` lets you walk sub‑folders (`rglob`).  
  * `--ext` forces you to look at any file types you produce.
* **`word_counter()`** – keeps reading all the text, tokenizing it (re‑adding word boundaries) and feeding them into a `collections.Counter`.
* **`top_k()`** – gets a pair of `(word, count)`.

Feel free to adapt this checklist or expand the module to include other corpora…
