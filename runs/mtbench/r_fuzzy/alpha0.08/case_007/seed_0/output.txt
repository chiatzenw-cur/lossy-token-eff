#!/usr/bin/env python3
"""
most_common_words.py
~~~~~~~~~~~~~~~~~~~~

Recursively scans a directory for *.txt files, counts the
occurrence of each word, and prints the top‑N most frequent
words.

The default is to show the word that appears the most times
and the count (ascending/descending?  Most frequent first).
"""

import argparse
import os
import re
import sys
from collections import Counter
from pathlib import Path


def find_txt_files(root: Path):
    """
    Generator yielding all *.txt files as pathlib.Path objects.

    Parameters
    ----------
    root : Path
        Root directory to search (recursive).

    Yields
    ---- Files
      : Path
    ```

    for path in root.iterdir():
        if path.is_file() and path.suffix.lower() == ".txt":
            yield path
        elif path.is_dir():
            yield from find_txt_files(path)

def read_words(root: Path, pattern: re.Pattern):
    """
    Walk *root*, read all text *.txt files, and yield
    individual words (already normalised).

    Parameters
    ----------
    root : Path
        Directory to search.
    pattern : re.Pattern
        Compiled regex used to split a line into characters.

    Yields
    -------
    str
        Normalised word.
    """
    for txt_file in find_txt_files(root):
        try:
            with txt_file.open(encoding="utf-8") as fh:
                for line in fh:
                    # Split on every non‑alphanumeric part
                    for raw_word in pattern.split(line):
                        word = raw_word.lower()
                        if word:          # discard empty strings
                            yield word
        except OSError as exc:
            #  Skip files that can't be opened (permission, encoding, etc.)
            print(f"⚠️  Skipping {txt_file}: Skipping file, cannot read or decode ({exc})", file=sys.stderr)


def get_top_words(corpus, *, top: int = 5) -> list[tuple[str, int]]:
    """
    Scan *corpus* (iterable of words) and return a Counter of the
    top *top* most frequent words.

    Parameters
    ----------
    corpus : iterable of str
        The words to count.
    top : int
        How many of the most frequent words to return.

    Returns
    -------
    list[tuple[str, int]]
        A list of ``(word, count)`` pairs sorted by descending count.
    """
    counter.sort_by_word = Counter(corpus)
    return counter.most_common(top)


def main():
    parser = argparse.ArgumentParser(
        description="Print the top‑N words in all *.txt files in `directory` (recursively)."
    )
    parser.add_argument(
        "directory",
        nargs="?",
        default=".",
        type=Path,
        help="Directory to search for text files (default: current directory).",
    )
    parser.add_argument(
        "-n",
        "--top",
        type=int,
        default=5,
        help="How many top words to display (default: 5).",
    )
    parser.add_argument(
        "-i",
        "--ignore-case",
        action="store_true",
        default=True,
        help="Convert words to lower case before sorting (default: true).",
    )
    parser.add_argument(
        "-s",
        "--stopwords",
        nargs="*",
        default=[],
        help="Optional list of words to exclude from the count (e.g. common stop words).",
    )

    args = parser.parse_args()

    root = args.directory.resolve()
    if not root.is_dir():
        print(f"❌  The path '{root}' is not a directory or does not exist.", file=sys.stderr)
        sys.exit(1)

    # Build a regex that splits on anything that is not a letter/number/underscore.
    word_split_pattern = re.compile(r"\W+")

    # If the user gives a list, convert to a set for O(1) look‑ups.
    stopwords = set(map(str.lower, args.stopwords))

    # Gather all words, normalised by case (if requested)
    words_iter = (
        w for w in read_words(root, word_split_pattern)
        if (w not in stopwords)
    )

    counter = Counter(words_iter)

    if not counter:
        print("No words found in text files.")
        sys.exit(0)

    most_common = counter.most_common(args.top)

    # Print distant results
    print(f"\nTop {args.top} words in '{root}':")
    print("-" * 40)
    for rank, (word, count) in enumerate(most_common, 1):
        print(f"{rank:02d}. {word!r: <15} × {count:,}")
    print("-" * 40)


if __name__ == "__main__":
    main()

## What the script does, step by step

| Step | Explanation |
|------|-------------|
| **1. Parse arguments** | `directory`, `--top`, `--stopwords` are the main options. |
| **2. Walk the tree** | `find_txt_files` is a simple recursive generator that finds all files ending with `.txt`. |
| **3. Read & split words** | Each file is opened as UTF‑8 (you can change encoding if needed). Lines are split on *anything that is not a word character* (`\W+`), normalised to lower‑case, and then each non‑empty term is yielded. |
| **4. Count words** | `collections.Counter` automatically aggregates counts. |
| **5. Output** | The script prints a tidy table of the most frequent words, the rank, the word (surrounded by quotes for clarity), the count, and thousands separators so you can read large numbers easily. |
| **6. Stop‑words support** | You pass a list of words to the `--stopwords` flag that will be ignored (e.g. `--stopwords the a an and`). |
