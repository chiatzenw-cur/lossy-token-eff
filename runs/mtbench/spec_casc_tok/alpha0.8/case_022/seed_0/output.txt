#include <iostream>
#include <limits>   // for std::numeric_limits
#include <cstdint>  // for int64_t

// ---------------------------------------------------------------------------
//  Recursive Fibonacci
// ---------------------------------------------------------------------------
// Returns the n-th Fibonacci number (1‑based indexing):
//
//   Fib(1) = 1
//   Fib(2) = 1
//   Fib(n) = Fib(n-1) + Fib(n-2)   for n > 2
//
// The function uses 64‑bit signed integers (int64_t).  For n > 92 the
// result does not fit in 64‑bit, so the program will warn the user
// for larger inputs.
int64_t fib(int n)
{
    if (n <= 0)   // defensive: we never call this with n <= 0
        return 0;
    if (n <= 2)   // base cases
        return 1;
    return fib(n - 1) + fib(n - 2);
}

// ---------------------------------------------------------------------------
//  Main
// ---------------------------------------------------------------------------
int main()
{
    std::cout << "Enter the position (n) of the Fibonacci number to compute: ";
    int n;
    std::cin >> n;

    // Basic input validation
    if (std::cin.fail() || n <= 0)
    {
        std::cerr << "Invalid input. Please enter a positive integer.\n";
        return 1;
    }

    // Warn the user if the result will overflow 64‑bit signed integer
    if (n > 92)
    {
        std::cerr << "Warning: The result will overflow a 64‑bit signed integer "
                  << "(max 9223372036854775807). Consider using a larger integer type.\n";
    }

    int64_t result = fib(n);
    std::cout << "Fib(" << n << ") = " << result << '\n';
    return 0;
}

1. **Input** – The program prompts the user for a positive integer `n`.
