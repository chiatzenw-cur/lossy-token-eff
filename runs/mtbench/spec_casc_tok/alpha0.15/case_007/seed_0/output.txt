1.  Walks through a user‑supplied directory (recursively) and opens every file that ends with `.txt`.
2.  Tokenises each file, normalises words to lower‑case and strips punctuation.
3.  Counts the frequency of every word using `collections.Counter`.
4.  Prints the **top‑5 most frequent words** together with their counts.

```python
#!/usr/bin/env python3
"""
word_counter.py

Read all *.txt files under a directory, count word frequencies,
and report the top‑5 most frequent words.
"""

import argparse
import os
import re
import sys
from collections import Counter
from typing import Iterable, Tuple

# --------------------------------------------------------------------------- #
# Helper functions
# --------------------------------------------------------------------------- #

WORD_RE = re.compile(r"\b\w+\b")  # matches sequences of alphanumerics


def words_from_text(text: str) -> Iterable[str]:
    """Yield all lower‑case words from *text*."""
    for match in WORD_RE.finditer(text):
        yield match.group(0).lower()


def words_from_file(path: str) -> Iterable[str]:
    """Yield words from the supplied file path."""
    try:
        with open(path, "r", encoding="utf-8") as fp:
            for line in fp:
                yield from words_from_text(line)
    except (IOError, OSError) as exc:
        print(f"⚠️  Could not read {path!r}: {exc}", file=sys.stderr)


def iterate_text_files(root: str) -> Iterable[str]:
    """Yield full paths of all *.txt files under *root* (recursively)."""
    for dirpath, _, filenames in os.walk(root):
        for fname in filenames:
            if fname.lower().endswith(".txt"):
                yield os.path.join(dirpath, fname)


# --------------------------------------------------------------------------- #
# Main logic
# --------------------------------------------------------------------------- #

def count_words_in_directory(root: str) -> Counter:
    """Return a Counter with word frequencies for all *.txt files in *root*."""
    counter = Counter()
    for txt_path in iterate_text_files(root):
        counter.update(words_from_file(txt_path))
    return counter


def main() -> None:
    parser = argparse.ArgumentParser(
        description="Count words in all *.txt files under a directory "
        "(recursively) and display the top‑5 most frequent words."
    )
    parser.add_argument(
        "directory",
        metavar="DIR",
        help="Root directory to search for *.txt files",
    )
    parser.add_argument(
        "-n",
        "--top",
        type=int,
        default=5,
        help="Number of top words to display (default: 5)",
    )
    args = parser.parse_args()

    if not os.path.isdir(args.directory):
        print(f"❌  {args.directory!r} is not a directory or cannot be accessed.", file=sys.stderr)
        sys.exit(1)

    counter = count_words_in_directory(args.directory)

    if not counter:
        print("⚠️  No words found in the provided directory.")
        sys.exit(0)

    print(f"\nTop {args.top} words in {args.directory!r}:\n")
    for word, count in counter.most_common(args.top):
        print(f"{word:>20} : {count}")

    # Optional: show total words counted
    print(f"\nTotal distinct words: {len(counter)}")
    print(f"Total words processed: {sum(counter.values())}")


if __name__ == "__main__":
    main()
```

## How to use

1. **Save** the script as `word_counter.py`.
2. **Make it executable** (optional):

   ```bash
   chmod +x word_counter.py
   ```

3. **Run** it, pointing to the directory you want to analyse:

   ```bash
   python3 word_counter.py /path/to/my/texts
   ```

   or, if you made it executable:

   ```bash

   ```

   * No arguments → program prints the top‑5 words.
