1. Walks through a *directory* (recursively).

```python
#!/usr/bin/env python3
"""
top-5-words.py

Given a directory, find all *.txt files underneath it and report the
five most frequently occurring words.
"""

import argparse
import collections
import os
import re
import sys


def find_txt_files(root: str):
    """
    Recursively walk *root* and yield absolute paths of all *.txt files.
    """
    for dirpath, _, filenames in os.walk(root):
        for fname in filenames:
            if fname.lower().endswith(".txt"):
                yield os.path.join(dirpath, fname)


def read_text(path: str) -> str:
    """
    Read a file with UTF‑8 encoded, ignoring decoding errors.
    """
    with open(path, encoding="utf-8", errors="ignore") as f:
        return f.read()


def tokenize(text: str):
    """
    Extract words from *text* (lower‑cased). Very simple: we split on
    word boundaries and strip everything that is not a letter/digit/pound
    (i.e. no ‘\w+’).
    """
    word_pattern = re.compile(r"\b\w+\b", re.UNICODE)
    return word_pattern.findall(text.lower())


def main():
    parser = argparse.ArgumentParser(
        description="Report the top‑5 most frequent words in all .txt files "
        "under a directory."
    )
    parser.add_argument(
        "directory",
        nargs="?",
        default=".",
        help="Directory to search (default: current directory)",
    )

    args = parser.parse_args()
    root_dir = os.path.abspath(args.directory)

    if not os.path.isdir(root_dir):
        print(f"Error: {root_dir!r} is not a directory.", file=sys.stderr)
        sys.exit(1)

    # Count all words in all files
    counter = collections.Counter()

    txt_files = list(find_txt_files(root_dir))
    if not txt_files:
        print("No .txt files found in the directory tree.")
        sys.exit(0)

    for filepath in txt_files:
        try:
            text = read_text(filepath)
            counter.update(tokenize(text))
        except Exception as exc:
            print(f"Warning: could not read {filepath!r}: {exc}", file=sys.stderr)

    # Show the top‑5 words
    print("\nTop‑5 words (by frequency):")
    for word, freq in counter.most_common(5):
        print(f"{word:<15} : {freq}")

    total = sum(counter.values())
    unique = len(counter)
    print(f"\nProcessed {len(txt_files)} file(s) with {total:,} unique unique words.")


if __name__ == "__main__":
    main()
```

## How it works

| Step | What happens | Why it matters |
|------|--------------|----------------|
| **`find_txt_files`** | Recursively walks the directory tree (`os.walk`) and yields paths that end in `.txt`. | Guarantees we only process text files, not images, binaries, etc. |
| **`read_text`** | Opens each file in UTF‑8, quietly ignoring decoding errors. | Most text files are UTF‑8; by ignoring errors we avoid crashing on stray bytes. |
| **`tokenize`** | Uses a regular expression to get word tokens (`\b\w+\b`). The lower‑case conversion makes the count case‑insensitive. | Keeps the script simple; for more elaborate tokenisation you could replace the regex. |
| **`Counter`** | `collections.Counter` aggregates word frequencies across the whole corpus. | Extremely fast native implementation. |
| **`most_common(5)`** | Retrieves the five most common words and their counts. | Simple, built‑in way to get the most frequent elements. |
| **Argument parsing** | `argparse` lets you specify the directory from the command line; otherwise it defaults to the current working directory. | Makes the script reusable and script‑friendly. |
