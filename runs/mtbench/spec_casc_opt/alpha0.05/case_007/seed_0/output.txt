import argparse
import os
import re
from collections import Counter
from typing import Dict, Iterable, List, Tuple

# --------------------------------------------------------------------------- #
# Utility functions
# --------------------------------------------------------------------------- #

def words_from_text(text: str) -> List[str]:
    """
    Split a text string into a list of words, stripped of punctuation and
    normalised to lower‑case.

    Parameters
    ----------
    text : str
        The raw text.

    Returns
    -------
    List[str]
        A list of cleaned words.
    """
    # Replace any non‑alphabetic characters with a space
    # (digits are not counted as words here; adjust if you want them)
    cleaned = re.sub(r"[^\w\s]", " ", text)
    # Split on whitespace and convert to lower‑case
    return [w.lower() for w in cleaned.split() if w]

def count_words_in_file(filepath: str) -> Counter:
    """
    Count words in a single text file.

    Parameters
    ----------
    filepath : str
        Path to the file.

    Returns
    -------
    collections.Counter
        Counter of word frequencies.
    """
    with open(filepath, 'r', encoding='utf-8') as f:
        text = f.read()
    return Counter(words_from_text(text))

def main_directory(directory: str, top_n: int = 5) -> None:
    """
    Walk through the directory, find all .txt files, count words,
    and yield the top‑n most frequent words.

    Parameters
    ----------
    directory : str
        Directory to search for.
    top_n : int
        Number of top words to display.
    """
    overall_totals = Counter()
    # os.walk yields dir, subdirs, filenames
    for root, _dirs, files in os.walk(directory):
        txt_files = [f for f in files if f.lower().endswith('.txt')]
        for fname in txt_files:
            full_path = os.path.join(root, fname)
            try:
                file_counts = count_words_in_file(full_path)
                overall_totals.update(file_counts)
            except Exception as exc:
                # If a file can't be read we cleanly skip it
                print(f"WARNING: Could not read {full_path!r}: {exc}")

    # Display the top‑n words
    print(f"\nTop {top_n} words in {directory!r}:\n")
    for rank, (word, count) in enumerate(overall_totals.most_common(top_n), start=1):

### How it works

| Step | What happens |
|------|--------------|
| **Argument parsing** | `argparse` lets you specify a directory (`-dir`) or a list of files (`-i`), and how many words to show (`-top`). |
| **Walking the directory** | `os.walk` walks the directory tree, collecting every file that ends with `.txt`. |
| **Counting words** | `count_words_in_file` reads a file, counts words with a `Counter`, and normalises punctuation and case using a regular expression. |
| **Aggregating** | The `Counter` objects from all files are combined with `update()`. |
| **Reporting** | `most_common` pulls the top‑`n` words from the combined counter and prints them. |
